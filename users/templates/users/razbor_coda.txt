Загрузка данных из Django (AJAX)
Этот блок выполняется один раз, когда страница полностью загружена в браузер.
document.addEventListener('DOMContentLoaded', function() { ... }): Это стандартная конструкция. 
Она говорит браузеру: «Когда HTML-страница будет полностью построена и дерево элементов (DOM) будет готово,
 выполни код, который находится внутри этой функции». 
Это гарантирует, что скрипт не попытается найти элементы на странице, которые ещё не загрузились.
            document.addEventListener('DOMContentLoaded', function() {
            // ... код внутри ...
            });
                ==================
fetch("{% url 'diary_data' %}")
fetch(...): Это современный способ отправить запрос на сервер, чтобы получить данные. 
В данном случае мы обращаемся к URL-адресу, который генерирует шаблонный тег 
Django {% url 'diary_data' %}. Э
тот адрес ведёт к функции diary_data в views.py, которая возвращает оценки студента.
                ===========================
.then(response => response.json())
.then(...): Так как серверу нужно время, чтобы ответить, fetch работает асинхронно. 
Метод .then() позволяет указать, что делать, 
когда ответ от сервера придёт.
response => response.json(): Мы берём ответ от сервера (response) и преобразуем его из формата 
JSON (текст) в формат JavaScript-объекта (данные), 
                =================
.then(data => {
    const tableBody = document.querySelector('#diary-table');
data => { ... }: Это следующий шаг. Он выполняется, когда данные успешно преобразованы в объект.
const tableBody = document.querySelector('#diary-table');: Мы находим в HTML-коде таблицу 
с id="diary-table" и сохраняем ссылку на неё в переменную tableBody, 
чтобы потом добавлять в неё строки.
        ======================
data.grades.forEach(item => {
      const row = document.createElement('tr');
      row.innerHTML = `
        <td>${item.date}</td>
        <td>${item.subject}</td>
        <td class="grade-cell" data-topic="${item.topic}">${item.value}</td>
      `;
      tableBody.appendChild(row);
    });
data.grades.forEach(...): Мы перебираем каждый элемент (оценку) в массиве grades, который пришёл от сервера.
const row = document.createElement('tr');: Для каждой оценки мы создаём новый элемент HTML — строку таблицы (<tr>).
row.innerHTML = ...: Мы заполняем эту строку тремя ячейками (<td>):
Дата: ${item.date}
Предмет: ${item.subject}
Оценка: ${item.value}. Эта ячейка имеет два важных атрибута:
class="grade-cell" — для последующей привязки события клика.
data-topic="${item.topic}" — в этом атрибуте "под капотом" хранится тема задания. 
Это стандартный способ хранить дополнительные данные в HTML.
tableBody.appendChild(row);: Готовую строку мы добавляем в таблицу на странице. 
В итоге таблица заполняется данными динамически, без перезагрузки страницы.
  ========================================
setupGradeClicks();
  });
});
setupGradeClicks();: После того как все строки добавлены в таблицу, мы вызываем функцию, 
которая "научит" ячейки с оценками реагировать на клики.

Эта функция ищет все созданные ячейки с оценками и вешает на них обработчик события.
function setupGradeClicks() {
  document.querySelectorAll('.grade-cell').forEach(cell => {
    cell.addEventListener('click', function() {
document.querySelectorAll('.grade-cell'): Находит все элементы на странице, 
у которых есть класс grade-cell. Это как раз те ячейки <td>, которые мы создали ранее.
.forEach(cell => { ... }): Перебирает каждую найденную ячейку.
cell.addEventListener('click', function() { ... }): Для каждой ячейки добавляет 
обработчик события "клик". Код внутри этой функции выполнится только тогда, 
когда пользователь нажмёт на эту конкретную ячейку.
        =========================
const topic = this.getAttribute('data-topic');
this.getAttribute('data-topic'): Внутри обработчика события this ссылается на тот элемент, 
по которому кликнули. Мы достаём из него значение атрибута data-topic, который мы записывали 
при создании строки. Теперь переменная topic содержит текст вроде "Интегралы" или 
"Великая Отечественная война".
        =====================================
document.getElementById('topic').innerText = 'Тема: ' + topic;
      document.getElementById('notification').style.display = 'block';
Находим на странице элемент с id="topic" (это тег <p> внутри блока уведомления) и меняем его текст 
    на "Тема: " + наша тема.
Находим блок уведомления (id="notification") и делаем его видимым, меняя CSS-свойство display на block.
        ====================================
setTimeout(() => {
        document.getElementById('notification').style.display = 'none';
      }, 3000);
    });
  });
}
setTimeout(...): Это таймер. Он говорит браузеру: «Подожди 3000 миллисекунд (3 секунды) и 
выполни код внутри».
Код внутри таймера снова находит блок уведомления и скрывает его (display = 'none').
Таким образом, при клике на оценку появляется всплывающее окно с темой, которое автоматически 
исчезает 
через 3 секунды.
